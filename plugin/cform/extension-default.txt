return static function(string $event, array &$context): void {

    $method = $event;

            // 每个表单生成管理列表行之前。
            // 可调整：$context中的vars。
            break;

        case 'form_list_render':
            // 表单管理列表整页渲染前，仅使用默认扩展。
            // 可调整：$context中的vars。
            break;

        case 'form_create_prepare':
            // 新建表单设计页：默认定义生成后、控件生成前。
            // 可调整：$context中的form。
            break;

        case 'form_create_render':
            // 新建表单设计页：展示变量生成后。
            // 可调整：$context中的vars。
            break;

        case 'form_update_prepare':
            // 编辑表单设计页：定义读取后、控件生成前。
            // 可调整：$context中的form。
            break;

            // 编辑表单设计页：展示变量生成后。
            // 可调整：$context中的vars。
            break;

        case 'form_create_validate':
            // 新增表单：基础定义验证后、入库前。
            // 可调整：$context中的form。
            break;

        case 'form_update_validate':
            // 修改表单：修订核对和基础验证后、入库前。
            // 可调整：$context中的form; previous为原定义。
            break;

        case 'form_create_persisted':
            // 新增表单已写入、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'form_created':
            // 新增表单已入库并取得编号。
            // 仅通知。
            break;

        case 'form_update_persisted':
            // 表单修改已写入、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'form_updated':
            // 修改表单已入库并更新修订号。
            // 仅通知。
            break;

        case 'form_delete_validate':
            // 表单停用前。
            // 抛出异常可阻止。
            break;

        case 'form_delete_persisted':
            // 停用标记已写入、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'form_deleted':
            // 表单已停用，报名记录仍保留。
            // 仅通知。
            break;

        case 'submission_prepare':
            // 前台填写页：初始数据准备后、生成控件前。
            // 可调整：$context中的data。
            break;

        case 'submission_render':
            // 前台填写页：展示变量生成后。
            // 可调整：$context中的vars; loops。
            break;

        case 'submission_input':
            // 前台提交：凭证核对及上传检查后、字段验证前。
            // 可调整：$context中的input。
            break;

        case 'submission_validate':
            // 前台提交：标准中文值验证后、入库前。
            // 可调整：$context中的data。
            break;

        case 'submission_persisted':
            // 前台新提交已写入并取得编号、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'submission_saved':
            // 前台新提交已入库并登记幂等结果。
            // 仅通知，重放不触发。
            break;

        case 'entry_create_prepare':
            // 后台新增记录：默认数据准备后、生成控件前。
            // 可调整：$context中的data。
            break;

        case 'entry_create_render':
            // 后台新增记录：展示变量生成后。
            // 可调整：$context中的vars; loops。
            break;

        case 'entry_update_prepare':
            // 后台编辑记录：记录读取后、生成控件前。
            // 可调整：$context中的data。
            break;

        case 'entry_update_render':
            // 后台编辑记录：展示变量生成后。
            // 可调整：$context中的vars; loops。
            break;

        case 'entry_create_input':
            // 后台新增记录：上传检查后、字段验证前。
            // 可调整：$context中的input。
            break;

        case 'entry_update_input':
            // 后台修改记录：旧记录读取及上传检查后、字段验证前。
            // 可调整：$context中的input。
            break;

        case 'entry_create_validate':
            // 后台新增记录：标准中文值验证后、入库前。
            // 可调整：$context中的data。
            break;

        case 'entry_update_validate':
            // 后台修改记录：标准中文值验证后、入库前。
            // 可调整：$context中的data; previous为原值。
            break;

        case 'entry_create_persisted':
            // 后台新增记录已写入并取得编号、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'entry_created':
            // 后台新增记录已入库并取得编号。
            // 仅通知。
            break;

        case 'entry_update_persisted':
            // 后台记录修改已写入、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'entry_updated':
            // 后台记录修改已入库。
            // 仅通知。
            break;

        case 'entry_list_query':
            // 后台记录列表：查询执行前。
            // 可调整：$context中的query。
            break;

        case 'entry_list_item':
            // 后台记录列表：每条记录生成单元格前。
            // 可调整：$context中的data。
            break;

        case 'entry_list_render':
            // 后台记录列表：整页展示变量生成后。
            // 可调整：$context中的vars; loops。
            break;

        case 'entry_delete_validate':
            // 记录归属核对后、删除前。
            // 抛出异常可阻止。
            break;

        case 'entry_delete_persisted':
            // 记录删除已执行、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'entry_deleted':
            // 记录已删除。
            // 仅通知。
            break;

        case 'public_list_query':
            // 公开列表或子块：公开权限核对后、查询前。
            // 可调整：$context中的query。
            break;

        case 'public_list_item':
            // 公开列表或子块：每条记录投影后、生成单元格前。
            // 可调整：$context中的data，仅公开字段。
            break;

        case 'public_block_render':
            // 公开列表子块：展示变量生成后。
            // 可调整：$context中的vars; loops。
            break;

        case 'public_list_render':
            // 公开列表页面：子块完成后、整页渲染前。
            // 可调整：$context中的vars; loops。
            break;

        case 'print_prepare':
            // 回执归属核对后、生成打印内容前。
            // 可调整：$context中的data。
            break;

        case 'print_render':
            // 打印页展示变量生成后。
            // 可调整：$context中的vars。
            break;

        case 'file_download_validate':
            // 附件权限、令牌及文件存在检查后，发送响应前。
            // 抛出异常可阻止。
            break;

        case 'import_prepare':
            // CSV上传与表头读取后，列名验证前。
            // 可调整：$context中的columns。
            break;

        case 'import_row_input':
            // 每行CSV拆列和多选解析后，字段验证前。
            // 可调整：$context中的input。
            break;

        case 'import_row_validate':
            // 每行CSV字段验证后，加入批次前。
            // 可调整：$context中的data。
            break;

        case 'import_validate':
            // 完整批次准备后，事务写入前。
            // 可调整：$context中的rows。
            break;

        case 'import_persisted':
            // 完整导入批次已写入、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'import_completed':
            // 整批导入事务成功后。
            // 仅通知，含ids。
            break;

        case 'export_prepare':
            // CSV生成前，尚未发送下载响应头。
            // 可调整：$context中的columns; query。
            break;

        case 'export_row':
            // 每条导出记录编码为CSV之前。
            // 可调整：$context中的data。
            break;

        case 'export_completed':
            // CSV完整生成后、发送响应前，不代表客户端已接收。
            // 仅通知。
            break;

        case 'mail_preview_prepare':
            // 邮件预览：记录读取后、模板生成前。
            // 可调整：$context中的data; recipient。
            break;

        case 'mail_preview_render':
            // 邮件预览：草稿生成后、预览页渲染前。
            // 可调整：$context中的mail.html; recipient。
            break;

        case 'mail_send_prepare':
            // 发送权领取和SMTP配置验证后、邮件生成前。
            // 可调整：$context中的data; recipient; input.mail_html。

        case 'mail_send_validate':
            // 实际邮件生成后、最终验证和SMTP调用前。
            // 可调整：$context中的mail.html; recipient。
            break;

        case 'mail_sent_persisted':
            // SMTP已确认、发送标记已写入但尚未提交。
            // 可调整：$context中的同一store事务联动，抛出异常保留待核对状态。
            break;

        case 'mail_sent':
            // SMTP确认成功且记录已标记发送后。
            // 仅通知。
            break;

        case 'mail_reset_validate':
            // 核对发送中状态后，重置发送状态前。
            // 抛出异常可阻止。
            break;

        case 'mail_reset_persisted':
            // 发送状态已重置、事务提交前。
            // 可调整：$context中的同一store事务联动，抛出异常回滚。
            break;

        case 'mail_reset':
            // 发送状态已恢复为未发送。
            // 仅通知。
            break;

        case 'operation_failed':
            // 已授权操作失败后，错误响应生成前。
            // 仅通知，含operation/error。
            break;

    }
};
